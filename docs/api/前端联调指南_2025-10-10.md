### 2. 配置请求拦截器

#### Axios示例 (React/Vue)

```javascript
// api/request.js
import axios from 'axios';

const request = axios.create({
  baseURL: 'http://localhost:8080/api/v1',
  timeout: 10000,
});

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    // 从localStorage获取token
    const token = localStorage.getItem('accessToken');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
request.interceptors.response.use(
  (response) => {
    const { code, data, message } = response.data;

    if (code === 200) {
      return data; // 直接返回data部分
    } else {
      // 错误处理
      console.error('API Error:', message);
      throw new Error(message);
    }
  },
  (error) => {
    // Token过期处理
    if (error.response?.status === 401) {
      // 跳转登录页
      localStorage.removeItem('accessToken');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default request;
```

#### Dio示例 (Flutter)

```dart
// lib/api/dio_client.dart
import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';

class DioClient {
  static final DioClient _instance = DioClient._internal();
  factory DioClient() => _instance;

  late Dio dio;

  DioClient._internal() {
    dio = Dio(BaseOptions(
      baseUrl: 'http://localhost:8080/api/v1',
      connectTimeout: Duration(seconds: 10),
      receiveTimeout: Duration(seconds: 10),
    ));

    // 请求拦截器
    dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        // 添加Token
        final prefs = await SharedPreferences.getInstance();
        final token = prefs.getString('accessToken');
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        return handler.next(options);
      },
      onResponse: (response, handler) {
        // 统一处理响应
        if (response.data['code'] == 200) {
          response.data = response.data['data'];
        }
        return handler.next(response);
      },
      onError: (error, handler) async {
        // Token过期处理
        if (error.response?.statusCode == 401) {
          final prefs = await SharedPreferences.getInstance();
          await prefs.remove('accessToken');
          // 跳转登录页
        }
        return handler.next(error);
      },
    ));
  }
}
```

## 认证流程

### 完整流程图

```
用户输入手机号
    ↓
发送验证码 (POST /auth/sms/send)
    ↓
用户输入验证码
    ↓
登录 (POST /auth/login/sms)
    ↓
保存 accessToken 和 refreshToken
    ↓
使用 accessToken 调用其他接口
    ↓
Token过期(2小时后)
    ↓
刷新Token (POST /auth/token/refresh)
```

### 代码示例

#### 1. 发送验证码

```javascript
// api/auth.js
import request from './request';

export const sendSmsCode = (mobile, type = 'login') => {
  return request.post('/auth/sms/send', {
    mobile,
    type  // ⚠️ 必须小写: login/register/reset
  });
};
```

**调用示例**:
```javascript
try {
  const result = await sendSmsCode('13800138000', 'login');
  console.log('短信ID:', result.smsId);
  console.log('过期时间:', result.expireTime, '秒');

  // 启动倒计时
  startCountdown(result.expireTime);
} catch (error) {
  console.error('发送验证码失败:', error.message);
}
```

#### 2. 验证码登录

```javascript
export const loginWithSms = (mobile, smsCode, smsId) => {
  return request.post('/auth/login/sms', {
    mobile,
    smsCode,
    smsId
  });
};
```

**调用示例**:
```javascript
try {
  const result = await loginWithSms('13800138000', '5379', 'sms_xxx');

  // 保存Token
  localStorage.setItem('accessToken', result.accessToken);
  localStorage.setItem('refreshToken', result.refreshToken);
  localStorage.setItem('userInfo', JSON.stringify({
    userId: result.userId,
    username: result.username,
    avatar: result.avatar
  }));

  // 跳转首页
  router.push('/');
} catch (error) {
  console.error('登录失败:', error.message);
}
```

#### 3. 刷新Token

```javascript
export const refreshToken = (refreshToken) => {
  return request.post('/auth/token/refresh', { refreshToken });
};
```

**使用场景**: 当接口返回401时自动刷新Token

```javascript
// 在响应拦截器中自动刷新
if (error.response?.status === 401) {
  try {
    const refreshToken = localStorage.getItem('refreshToken');
    const result = await refreshToken(refreshToken);

    // 更新Token
    localStorage.setItem('accessToken', result.accessToken);

    // 重试原请求
    return request(error.config);
  } catch (e) {
    // 刷新失败,跳转登录
    localStorage.clear();
    window.location.href = '/login';
  }
}
```

## 常见问题FAQ

### Q1: 为什么商品ID有时是String,有时是Number?

**A**: 不同接口返回的ID类型可能不同:
- 商品列表接口: String类型 `"1969761981678358528"`
- 搜索接口: Number类型 `1969761981678358528`

**建议**: 前端统一转换为String处理,避免精度丢失

```javascript
const productId = String(response.id);
```

### Q2: 分页数据在哪个字段?

**A**: 大部分接口使用 `content` 字段:

```javascript
// ✅ 正确
const products = response.data.content;
const pagination = response.data.pagination;

// ❌ 错误
const products = response.data.products;  // undefined
```

**例外**: 交易列表接口使用 `transactions` 字段

### Q3: Token过期怎么处理?

**A**: 有两种方案:

**方案1: 自动刷新**
```javascript
// 在响应拦截器中自动刷新Token
if (error.response?.status === 401) {
  const newToken = await refreshToken(oldRefreshToken);
  // 重试原请求
}
```

**方案2: 跳转登录**
```javascript
if (error.response?.status === 401) {
  localStorage.clear();
  router.push('/login');
}
```

### Q4: 如何处理中文乱码?

**A**: 后端已统一使用UTF-8编码,前端无需特殊处理:

```javascript
// 直接使用中文
const result = await sendMessage(chatRoomId, {
  type: 'TEXT',
  content: '你好,这是中文消息'  // ✅ 直接发送
});
```

### Q5: 交易ID格式是什么?

**A**: 交易ID格式为 `tx_{数字ID}`:
- 完整格式: `"tx_1976621912847159296"`
- 可以使用完整格式或纯数字ID调用API

```javascript
// 两种方式都可以
await getTransactionDetail('tx_1976621912847159296');
await getTransactionDetail('1976621912847159296');
```

### Q6: 如何获取验证码?(测试环境)

**A**: 开发环境下验证码会打印在后端日志中:

1. 调用发送验证码接口
2. 查看后端控制台日志
3. 搜索 "Your verification code is"

```bash
# 查看日志
tail -f ~/Desktop/second-hand-platform2/app.log | grep "verification code"
```

### Q7: 聊天室的activeTransactions是什么?

**A**: 这是新增的字段,表示该聊天室中所有进行中的交易:

```javascript

  console.log('我的角色:', tx.role);             // BUYER/SELLER

  // 只有AGREED状态且是买家时才有交易码
```

## 错误处理

### 统一错误码

| 错误码 | 说明 | 处理建议 |
|--------|------|----------|
| 200 | 成功 | - |
| 400 | 请求参数错误 | 检查请求参数 |
| 401 | 未授权 | Token过期,重新登录 |
| 403 | 无权限 | 提示用户无权限操作 |
| 404 | 资源不存在 | 提示资源不存在 |
| 429 | 请求过于频繁 | 提示用户稍后再试 |
| 500 | 服务器错误 | 提示系统错误,稍后再试 |

### 错误处理示例

```javascript
try {
  const result = await api.someMethod();
  // 处理成功
} catch (error) {
  if (error.response) {
    // 服务器返回错误
    const { status, data } = error.response;

    switch (status) {
      case 401:
        // Token过期
        localStorage.clear();
        router.push('/login');
        break;

      case 429:
        // 请求过于频繁
        message.error('操作过于频繁,请稍后再试');
        break;

      case 500:
        // 服务器错误
        message.error(data.message || '系统错误,请稍后再试');
        break;

      default:
        message.error(data.message || '请求失败');
    }
  } else {
    // 网络错误
    message.error('网络连接失败,请检查网络');
  }
}
```

### 1. Token管理

```javascript
// utils/formatter.js

// 格式化价格
export const formatPrice = (price) => {
  return `¥${price.toLocaleString()}`;
};

// 格式化时间
export const formatTime = (timeStr) => {
  const now = new Date();
  const time = new Date(timeStr);
  const diff = now - time;

  const minute = 60 * 1000;
  const hour = 60 * minute;
  const day = 24 * hour;

  if (diff < minute) return '刚刚';
  if (diff < hour) return `${Math.floor(diff / minute)}分钟前`;
  if (diff < day) return `${Math.floor(diff / hour)}小时前`;
  if (diff < 7 * day) return `${Math.floor(diff / day)}天前`;

  return timeStr.split(' ')[0];  // 返回日期部分
};

// 格式化手机号
export const formatMobile = (mobile) => {
  if (!mobile) return '';
  return mobile.replace(/(\d{3})\d{4}(\d{4})/, '$1****$2');
};

// 格式化交易ID
export const formatTransactionId = (id) => {
  if (typeof id === 'string' && id.startsWith('tx_')) {
    return id;
  }
  return `tx_${id}`;
};
```
